\section*{Q2: BM25 Lexical Retrieval}

\textbf{Method}: Inverted index on titles/abstracts. Candidate-restricted scoring: $O(\bar{c})$ vs $O(N)$ ($10-100\times$ speedup).

\textbf{Formula}: $\sum_{i} \text{IDF}(q_i) \cdot \frac{f(q_i,d) \cdot (k_1+1)}{f(q_i,d) + k_1(1-b+b \cdot |d|/L)}$ with $k_1=1.5, b=0.75$

\section*{Codabench: Two-Stage Pipeline}

\textbf{Stage 1}: Precompute BM25 + semantic scores for 13.5M impressions ($\sim$3--4 hours, batch=10K impressions)

\textbf{Stage 2}: Weighted fusion $\text{score}_{\text{fused}} = (1-\alpha) \cdot \text{norm}(\text{BM25}) + \alpha \cdot \text{norm}(\text{semantic})$ ($\sim$1--2 min per alpha value)

\textbf{Efficiency}: 1 alpha = 4h, 9 alphas = 4.25h (vs 36--40h naive). Status: Stage 1 at 27.9\% (3.77M impressions processed)

\section*{Design Choices}

\subsection*{1. BM25: Candidate-Restricted Scoring}
Score only impression candidates ($O(\bar{c}) \approx 9$) vs full corpus ($O(N) = 125.5K$). 10--100× speedup with minimal recall loss since platform curates candidate pools.

\subsection*{2. Semantic: Multilingual Embeddings}
EB-NeRD articles in Danish (``Hanks beskyldt for mishandling''). Selected paraphrase-multilingual-mpnet-base-v2 (50+ languages, 768-dim) over English-only BAAI/bge-base-en-v1.5. MIND keeps English model.

\subsection*{3. ANN Index: FAISS IVF}
Compared LSH (memory-heavy), HNSW (slower indexing), Product Quant (worse speed). IVF balances $O(n)$ build, $O(k + m \times d)$ query, 90--98\% recall @ nprobe=10.

\subsection*{4. Evaluation: Per-Impression}
Per-impression (4K--30K samples) vs user-aggregated (2K--800 samples). Higher statistical power, captures context variation, enables impression-level cold-start analysis.

\subsection*{5. Two-Stage Pipeline}
Kaggle timeout ($\sim$9h). Precompute all scores ($\sim$4h) + apply alpha in 1--2 min. Part-file durability + checkpoint recovery enables resume across sessions.

\section*{Key Findings}

\begin{enumerate}
    \item \textbf{EB-NeRD}: Semantic outperforms BM25 (+4.7\% nDCG@10, +7.3\% MRR) with perfect novelty (100\%) and 99.7\% catalog coverage. Semantic exploration beats click-pattern fitting on small catalogs.

    \item \textbf{MIND}: Semantic gains (+3.8\% nDCG@10, +3.5\% MRR, +1.2\% Recall@50, 99.94\% Recall@200). Cold-start users show 22\% higher AUC + 9.2\% higher diversity; warm users show ranking gains.

    \item \textbf{Multilingual Model Critical}: Danish EB-NeRD requires paraphrase-multilingual-mpnet-base-v2. English-only baseline would significantly degrade semantic signal.

    \item \textbf{FAISS IVF Effective}: 100 clusters, nprobe=10 achieves Recall@50 $\geq$ 0.92, Recall@200 $\geq$ 0.99 across both datasets. Avoids brute-force $O(n \times d)$ complexity.

    \item \textbf{Coverage-Exploration Tradeoff}: EB-NeRD semantic explores (99.7\% vs 100\% BM25). MIND semantic broader (86\% vs 87.3\%), reflects dataset size differences.
\end{enumerate}
